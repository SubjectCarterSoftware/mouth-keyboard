<interfaces>
Keep the production flow aligned with the existing trigger APIs and introduce the thinnest new testable seam for calibration capture:

```swift
@MainActor
final class ShellPreferences: ObservableObject {
    @Published private(set) var activeTriggerProfile: TriggerProfile
    func setTriggerPreset(_ preset: TriggerNamePreset)
    func setCustomTrigger(primary: String, aliases: [String])
    func applyCalibrationAliases(_ aliases: [String])
}
```

```swift
struct TriggerCalibrationSession {
    mutating func recordSample(_ sample: CalibrationSample?) -> CalibrationStepResult
    var isComplete: Bool { get }
    func finalizedAliases() -> [String]
}
```

```swift
protocol CalibrationSampleCapturing: Sendable {
    func captureSample(for primaryName: String) async throws -> CalibrationSample?
}
```

The executor may name the protocol differently, but the calibration runner must be independently testable and must not depend on the global ActivationStore recording state machine.
</interfaces>
