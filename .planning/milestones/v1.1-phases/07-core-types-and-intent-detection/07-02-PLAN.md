**Algorithm (per RESEARCH.md pattern 3 with all documented pitfall fixes applied):**

```swift
import Foundation

/// Caseless enum used as a namespace for pure static detection logic.
///
/// `detect(transcript:modes:)` inspects the transcript for a leading or trailing
/// trigger phrase and returns a `ConvertIntent`. If both are present, the trailing
/// trigger wins (per spec). Detection is case-insensitive. The original transcript
/// is always preserved on `ConvertIntent.originalTranscript`.
enum IntentDetector {

    /// Detects a ConvertMode trigger phrase at the start or end of `transcript`.
    ///
    /// - Parameters:
    ///   - transcript: The raw Whisper transcript string (may have leading/trailing whitespace).
    ///   - modes: The modes to match against. Pass `ConvertMode.allCases` for built-in modes.
    ///            `.passthrough` entries are automatically skipped.
    /// - Returns: A `ConvertIntent`. When no trigger is detected, `mode` is `.passthrough`
    ///            and `strippedBody == originalTranscript`.
    static func detect(transcript: String, modes: [ConvertMode]) -> ConvertIntent {
        // Normalize: trim outer whitespace, lowercase for matching.
        // Use the trimmed original (preserving inner casing) for body extraction.
        let trimmedOriginal = transcript.trimmingCharacters(in: .whitespacesAndNewlines)
        let normalized = trimmedOriginal.lowercased()

        // For trailing-suffix check: also try stripping one terminal punctuation character
        // (.  ,  !  ?) since Whisper inserts a period after utterances (Pitfall 2).
        let terminalPunctuation: Set<Character> = [".", ",", "!", "?"]
        let normalizedStripped: String
        if let last = normalized.last, terminalPunctuation.contains(last) {
            normalizedStripped = String(normalized.dropLast())
        } else {
            normalizedStripped = normalized
        }

        var leadingResult: ConvertIntent? = nil
        var trailingResult: ConvertIntent? = nil

        for mode in modes {
            // Skip .passthrough — its defaultActivationPhrase is "" and an empty
            // hasSuffix/hasPrefix would match everything (Pitfall 5).
            guard mode != .passthrough else { continue }

            // Match against phrase candidates supplied by the mode contract itself.
            // In Phase 7, built-in modes expose the four locked trigger variants via
            // ConvertMode.activationPhraseCandidates. Future phases can extend the set
            // of passed modes without changing IntentDetector's control flow.
            for candidate in mode.activationPhraseCandidates {
                let candidateLength = candidate.count  // ASCII-safe; Character count == byte count

                // Leading check against `normalized` (full trimmed normalized transcript)
                if normalized.hasPrefix(candidate) {
                    let bodyStart = trimmedOriginal.index(
                        trimmedOriginal.startIndex,
                        offsetBy: candidateLength
                    )
                    let rawBody = String(trimmedOriginal[bodyStart...])
                        .trimmingCharacters(in: .whitespacesAndNewlines)
                    leadingResult = ConvertIntent(
                        mode: mode,
                        strippedBody: rawBody,
                        originalTranscript: transcript
                    )
                }

                // Trailing check — try both normalized and punctuation-stripped variants
                for normalizedVariant in [normalized, normalizedStripped] {
                    if normalizedVariant.hasSuffix(candidate) {
                        // Extract body from the original trimmed string.
                        // The candidate length is the same in both normalized variants.
                        let usedLength: Int
                        if normalizedVariant == normalizedStripped && normalizedStripped.count < normalized.count {
                            // Terminal punctuation was stripped; body ends one character earlier
                            usedLength = candidateLength + (normalized.count - normalizedStripped.count)
                        } else {
                            usedLength = candidateLength
                        }
                        let bodyEnd = trimmedOriginal.index(
                            trimmedOriginal.endIndex,
                            offsetBy: -usedLength
                        )
                        let rawBody = String(trimmedOriginal[..<bodyEnd])
                            .trimmingCharacters(in: .whitespacesAndNewlines)
                        trailingResult = ConvertIntent(
                            mode: mode,
                            strippedBody: rawBody,
                            originalTranscript: transcript
                        )
                        break  // Don't double-assign trailingResult for the same mode
                    }
                }
            }
        }

        // End-wins: trailing result takes priority over leading result (per spec).
        return trailingResult
            ?? leadingResult
            ?? ConvertIntent(mode: .passthrough, strippedBody: transcript, originalTranscript: transcript)
    }
}
```

**Implementation notes (pitfall guards already embedded above):**
- `ConvertMode.activationPhraseCandidates` is ordered longest-first, so `"convert to"` / `"format to"` win before `"convert"` / `"format"` (Pitfall 3)
- `detect()` consumes the mode-owned phrase contract directly, so it does not hard-code a "strip convert to" assumption
- `dropFirst`/body extraction always operates on `trimmedOriginal` (correct casing), not the lowercased copy (Pitfall 4)
- `guard mode != .passthrough` prevents empty-phrase suffix matching (Pitfall 5)
- Whitespace trimmed from `transcript` before `normalized` is built (Pitfall 1)
- Terminal punctuation stripped for suffix check (Pitfall 2)
- `leadingResult` may be set but is overridden by `trailingResult` (end-wins, Pitfall 6 of single-pass)
