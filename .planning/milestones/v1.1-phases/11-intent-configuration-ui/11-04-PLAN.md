    **IntentEditViewModel:**
    ```swift
    @MainActor
    final class IntentEditViewModel: ObservableObject {
        @Published var modeName: String = ""
        @Published var systemPrompt: String = ""
        @Published var suggestedName: String = ""
        @Published var previewOutput: String = ""
        @Published var isGeneratingPreview: Bool = false
        @Published var phraseTesterInput: String = ""
        @Published var phraseTesterResult: String = ""   // "Triggered" / "Not triggered (confidence: 0.XX)"

        let isBuiltIn: Bool
        let originalMode: ConvertMode?   // non-nil for built-ins; nil for new custom
        let entryID: String              // ConvertMode.rawValue or UUID string

        private var cancellables = Set<AnyCancellable>()
        private var previewTask: Task<Void, Never>?
        private var phraseGenTask: Task<Void, Never>?

        static let exampleInput = "had a call with the team today we covered the roadmap and need to follow up with Sarah by Friday"
    }
    ```

    Combine pipelines in `init`:
    - `$systemPrompt.debounce(for: .seconds(2), scheduler: RunLoop.main)` → `triggerLivePreview(prompt:)` and `generatePhrasePatterns(prompt:)`
    - `$systemPrompt.debounce(for: .seconds(1), scheduler: RunLoop.main)` → `suggestName(prompt:)`
    - `$phraseTesterInput.debounce(for: .seconds(0.5), scheduler: RunLoop.main)` → `runPhraseTester(input:)`

    `triggerLivePreview(prompt:)`:
    - Cancel previous `previewTask`
    - Set `isGeneratingPreview = true`
    - `Task { rewritten = try await LLMRewriteService.shared.rewrite(body: Self.exampleInput, instructions: prompt); previewOutput = rewritten; isGeneratingPreview = false }`
    - On error: `previewOutput = "[Preview failed]"; isGeneratingPreview = false`

    `generatePhrasePatterns(prompt:)`:
    - Cancel previous `phraseGenTask`
    - Generation prompt: `"Generate 50 natural spoken phrases a person might use to activate this mode. One phrase per line. Lowercase only. No numbers, no punctuation except spaces. No commentary or explanations.\n\nMode description: \(prompt)"`
    - Call `LLMRewriteService.shared.rewrite(body: "", instructions: generationPrompt)`
    - Parse: `output.components(separatedBy: "\n").map { $0.trimmingCharacters(in: .whitespacesAndNewlines) }.filter { !$0.isEmpty }.prefix(50)`
    - Save to `UserIntentStore`: update `phrasePatterns` on existing entry. If no entry exists yet, wait for user to save.
    - Silent: no `@Published` property for this, no UI update.

    `suggestName(prompt:)`:
    - Name suggestion prompt: `"In 2-4 words, give a concise name for a dictation mode that does the following. Reply with only the name, no explanation.\n\nMode description: \(prompt)"`
    - `suggestedName = (try? await LLMRewriteService.shared.rewrite(body: "", instructions: suggestionPrompt))?.trimmingCharacters(in: .whitespacesAndNewlines) ?? ""`

    `runPhraseTester(input:)`:
    - Snapshot store entries synchronously (use a local `@Published var cachedEntries` populated on appear)
    - Call `IntentDetector.detect(transcript: input, definitions: IntentCatalog.effective(store: cachedEntries))`
    - Format result: if `intent.mode != .passthrough || intent.customIntentID != nil` → `"Triggered as \(matchedName)"` else `"Not triggered"`

    `save()` async func (called by view on Save button):
    - For built-in: `try await UserIntentStore.shared.addOrUpdateBuiltInOverride(entry)`
    - For custom (new): `try await UserIntentStore.shared.addOrUpdateCustomMode(entry)`
    - For custom (edit): same `addOrUpdateCustomMode`

    `reset()` async func (built-in only):
    - `try await UserIntentStore.shared.resetBuiltIn(mode: originalMode!)`
    - Reset local fields to `originalMode.defaultSystemPrompt`

    `delete()` async func (custom only):
    - `try await UserIntentStore.shared.deleteCustomMode(id: entryID)`

    **IntentEditView layout (SwiftUI):**
    ```
    HSplitView {
        // Left: Definition (min width 280)
        VStack(alignment: .leading, spacing: 16) {
            // Mode Name
            LabeledContent("Mode Name") {
                TextField("", text: $vm.modeName,
                          prompt: Text(vm.suggestedName).foregroundStyle(.tertiary))
            }

            // System Prompt
            LabeledContent("Prompt") {
                TextEditor(text: $vm.systemPrompt)
                    .frame(minHeight: 120)
                    .overlay(RoundedRectangle(cornerRadius: 4).stroke(Color.secondary.opacity(0.3)))
            }
            .help("Describe what this mode should do with your dictated text...")

            // Actions row
            HStack {
                if vm.isBuiltIn {
                    Button("Reset to Default") { Task { await vm.reset() } }
                } else {
                    Button("Delete", role: .destructive) { Task { await vm.delete() } }
                }
                Spacer()
                Button("Save") { Task { await vm.save() } }
                    .buttonStyle(.borderedProminent)
                    .disabled(vm.modeName.isEmpty || vm.systemPrompt.isEmpty)
            }

            // Phrase trigger tester (collapsed by default)
            DisclosureGroup("Test trigger phrase") {
                VStack(alignment: .leading, spacing: 8) {
                    TextField("Type how you'd say this out loud...", text: $vm.phraseTesterInput)
                    if !vm.phraseTesterResult.isEmpty {
                        Text(vm.phraseTesterResult)
                            .font(.caption)
                            .foregroundStyle(.secondary)
                    }
                }
                .padding(.top, 4)
            }
        }
        .padding()
        .frame(minWidth: 280, idealWidth: 320)

        // Right: Live Preview (min width 300)
        VStack(alignment: .leading, spacing: 8) {
            Text("Live Preview").font(.headline)
            if vm.isGeneratingPreview {
                ProgressView().frame(maxWidth: .infinity, alignment: .center)
            } else {
                ScrollView {
                    Text(vm.previewOutput.isEmpty ? "Preview will appear after you type a system prompt..." : vm.previewOutput)
                        .font(.body)
                        .foregroundStyle(vm.previewOutput.isEmpty ? .secondary : .primary)
                        .frame(maxWidth: .infinity, alignment: .leading)
                }
                .frame(minHeight: 120)
                .padding(8)
                .background(Color(nsColor: .textBackgroundColor))
                .clipShape(RoundedRectangle(cornerRadius: 4))
            }
            Text("Input: \"\(IntentEditViewModel.exampleInput)\"")
                .font(.caption)
                .foregroundStyle(.secondary)
        }
        .padding()
        .frame(minWidth: 300, idealWidth: 380)
    }
    ```
